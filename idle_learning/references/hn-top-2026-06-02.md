## Fetch Script Used

# /tmp/hn_top_20260602.py — timestamped, cron-safe
import urllib.request, json
url = "https://hacker-news.firebaseio.com/v0/topstories.json"
req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
with urllib.request.urlopen(req, timeout=10) as resp:
    ids = json.loads(resp.read())
for i, story_id in enumerate(ids[:10]):
    story_url = f"https://hacker-news.firebaseio.com/v0/item/{story_id}.json"
    try:
        req2 = urllib.request.Request(story_url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req2, timeout=6) as r:
            story = json.loads(r.read())
            print(f"{i+1}. [{story.get('score',0)}pts] {story.get('title','')}")
    except Exception as e:
        print(f"Error: {e}")
